# Traiter toutes les factures clients avec TVA automatique
factures = [
    {"client": "SONIBANK",     "montant_ht": 500_000},
    {"client": "MOOV AFRICA",  "montant_ht": 1_200_000},
    {"client": "SOBEBRA",     "montant_ht": 750_000},
    {"client": "CAJAF",       "montant_ht": 320_000},
]

print(f"{'Client':20} {'HT':>12} {'TVA':>12} {'TTC':>12}")
print("-" * 58)

for f in factures:
    ht  = f["montant_ht"]
    tva = ht * 0.18      # TVA Bénin 18%
    ttc = ht + tva
    print(f"{f['client']:20} {ht:>12,.0f} {tva:>12,.0f} {ttc:>12,.0f}")
