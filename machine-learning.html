# ─── Prédiction rendement agricole — Bénin ───
from sklearn.pipeline import Pipeline
from sklearn.ensemble import GradientBoostingRegressor
from sklearn.model_selection import cross_val_score
from sklearn.metrics import mean_absolute_error, r2_score

# Pipeline : préprocessing + Gradient Boosting
pipeline = Pipeline([
    ('preprocessing', preprocesseur),
    ('model', GradientBoostingRegressor(n_estimators=200, random_state=42))
])

# Cross-validation 5-fold
scores = cross_val_score(pipeline, X_train, y_train, cv=5, scoring='r2')
print(f"R² moyen : {scores.mean():.3f} ± {scores.std():.3f}")

# Prédire une nouvelle exploitation (Borgou, maïs)
pipeline.fit(X_train, y_train)
prediction = pipeline.predict(nouvelle_exploitation)
print(f"Rendement prédit : {prediction[0]:.0f} kg/ha")
